/**
 * Inisialisasi awal sistem (WAJIB DIJALANKAN SEKALI)
 */
function initSetup() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Create Sheets if not exist
  Object.values(SHEETS).forEach(sheetName => {
    if (!ss.getSheetByName(sheetName)) {
      ss.insertSheet(sheetName);
    }
  });

  // 2. Setup Headers
  const headers = {
    [SHEETS.SISWA]: ['ID_SISWA', 'NAMA', 'KELAS', 'JENJANG', 'WALI', 'WA', 'FOTO', 'CREATED_AT'],
    [SHEETS.ABSENSI]: ['ID_SISWA', 'TANGGAL', 'JAM', 'STATUS', 'KETERANGAN'],
    [SHEETS.USERS]: ['EMAIL', 'PASSWORD', 'ROLE', 'NAMA'],
    [SHEETS.LOG_WA]: ['ID_SISWA', 'PESAN', 'STATUS', 'TIMESTAMP'],
    [SHEETS.SETTING]: ['KEY', 'VALUE']
  };

  Object.entries(headers).forEach(([sheetName, headerRow]) => {
    const sheet = ss.getSheetByName(sheetName);
    sheet.getRange(1, 1, 1, headerRow.length).setValues([headerRow]).setFontWeight('bold');
  });

  // 3. Fill Dummy Data
  fillDummyData();
}

/**
 * API: Login
 */
function login(email, password) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEETS.USERS);
  const data = sheet.getDataRange().getValues();
  
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === email && data[i][1] === password) {
      return {
        success: true,
        user: {
          email: data[i][0],
          role: data[i][2],
          nama: data[i][3]
        }
      };
    }
  }
  return { success: false, message: 'Email atau Password salah' };
}

/**
 * API: Get All Students
 */
function getSiswa() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEETS.SISWA);
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const result = [];
  
  for (let i = 1; i < data.length; i++) {
    const obj = {};
    headers.forEach((header, index) => {
      obj[header] = data[i][index];
    });
    result.push(obj);
  }
  return result;
}

/**
 * API: Submit Attendance
 */
function submitAbsensi(idSiswa, status, keterangan = '') {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const absensiSheet = ss.getSheetByName(SHEETS.ABSENSI);
  const siswaSheet = ss.getSheetByName(SHEETS.SISWA);
  
  // 1. Validate Student
  const siswaData = siswaSheet.getDataRange().getValues();
  let siswa = null;
  for (let i = 1; i < siswaData.length; i++) {
    if (siswaData[i][0] === idSiswa) {
      siswa = { id: siswaData[i][0], nama: siswaData[i][1], wa: siswaData[i][5] };
      break;
    }
  }
  
  if (!siswa) return { success: false, message: 'Siswa tidak ditemukan' };
  
  // 2. Anti-Double Scan (1x per day)
  const today = Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd");
  const absensiData = absensiSheet.getDataRange().getValues();
  for (let i = 1; i < absensiData.length; i++) {
    const rowDate = Utilities.formatDate(new Date(absensiData[i][1]), "GMT+7", "yyyy-MM-dd");
    if (absensiData[i][0] === idSiswa && rowDate === today) {
      return { success: false, message: 'Sudah absen hari ini' };
    }
  }
  
  // 3. Save Attendance
  const now = new Date();
  const jam = Utilities.formatDate(now, "GMT+7", "HH:mm:ss");
  absensiSheet.appendRow([idSiswa, now, jam, status, keterangan]);
  
  // 4. (Optional) Trigger WA Notification Logic here
  
  return { success: true, message: `Absensi ${siswa.nama} berhasil`, data: { nama: siswa.nama, jam: jam } };
}

/**
 * API: Get Dashboard Stats
 */
function getDashboardStats() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const siswaSheet = ss.getSheetByName(SHEETS.SISWA);
  const absensiSheet = ss.getSheetByName(SHEETS.ABSENSI);
  
  const totalSiswa = siswaSheet.getLastRow() - 1;
  const today = Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd");
  const absensiData = absensiSheet.getDataRange().getValues();
  
  let hadirToday = 0;
  let terlambatToday = 0;
  
  for (let i = 1; i < absensiData.length; i++) {
    const rowDate = Utilities.formatDate(new Date(absensiData[i][1]), "GMT+7", "yyyy-MM-dd");
    if (rowDate === today) {
      if (absensiData[i][3] === 'HADIR') hadirToday++;
      if (absensiData[i][3] === 'TERLAMBAT') terlambatToday++;
    }
  }
  
  return {
    totalSiswa,
    hadirToday,
    terlambatToday,
    tidakHadirToday: totalSiswa - (hadirToday + terlambatToday)
  };
}
